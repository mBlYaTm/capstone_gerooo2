  Object.values(DET_COLORS).forEach(({id,cls})=>{

  });

  // Fire detectors based on reasons
  reasons.forEach(reason=>{

    if(DET_COLORS[key]){
      const el=document.getElementById(DET_COLORS[key].id);
      el.classList.add(DET_COLORS[key].cls);

    Object.values(DET_COLORS).forEach(({id,cls})=>document.getElementById(id).classList.add(cls));

function connectSSE() {
  const es = new EventSource(`${API}/events`);

    try {

      } else if (msg.type === 'log') {

        addTelegramAlert(d.event, d.ip, d.score, d.reasons);
        log(`🚨 HIGH RISK — ${d.reasons.join(', ')} — score: ${d.score}`, 'err');

        const d = msg.data;
        updateRisk(d.score, d.level, d.reasons);
      }
    } catch(err) {}
  };

  es.onerror = function() {
    setTimeout(connectSSE, 3000);
  };
}

  const bar=document.getElementById('risk-bar');

  bar.style.width=Math.min(score,100)+'%';

  bar.style.background=map[level]||'var(--green)';
  lbl.style.color=map[level]||'var(--green)';

  // Reset all detectors first
  Object.values(DET_COLORS).forEach(({id,cls})=>{
    document.getElementById(id).classList.remove('fired-red','fired-yellow','fired-purple','fired-cyan');
  });

  // Fire detectors based on reasons
  reasons.forEach(reason=>{
    const key=reason.toLowerCase().replace(/ /g,'_');
    if(DET_COLORS[key]){
      const el=document.getElementById(DET_COLORS[key].id);
      el.classList.add(DET_COLORS[key].cls);
    }
  });

  // If HIGH — fire all
  if(level==='HIGH'&&reasons.length===0){
    Object.values(DET_COLORS).forEach(({id,cls})=>document.getElementById(id).classList.add(cls));
  }
}

function connectSSE() {
  const es = new EventSource(`${API}/events`);

  es.onmessage = function(e) {
    try {
      const msg = JSON.parse(e.data);
      if (msg.type === 'connected') {
        log('Dashboard connected to server · real-time active', 'inf');
      } else if (msg.type === 'log') {
        log(msg.data.msg, msg.data.level || 'inf');
      } else if (msg.type === 'telegram_alert') {
        const d = msg.data;
        addTelegramAlert(d.event, d.ip, d.score, d.reasons);
        log(`🚨 HIGH RISK — ${d.reasons.join(', ')} — score: ${d.score}`, 'err');
      } else if (msg.type === 'detector_fired') {
        const d = msg.data;
        updateRisk(d.score, d.level, d.reasons);
      }
    } catch(err) {}
  };

  es.onerror = function() {
    setTimeout(connectSSE, 3000);
  };
}

function connectSSE() {

  };

  es.onerror = function() {

// VALIDATE
async function validateToken(){
  const token=document.getElementById('val-tok').value.trim();
  const out=document.getElementById('out-validate');
  if(!token){out.textContent='Paste a token first';return;}
  out.className='out-box';out.textContent='Validating…';
  try{
    const r=await fetch(`${API}/validate`,{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({token})});
    const d=await r.json();
    reqCount++;animate('s-req',reqCount);
    if(r.ok){
      out.className='out-box ok';
      out.textContent=`✓ Token VALID\nUses left: ${d.uses_left}\nRisk: ${d.risk?.level||'LOW'} (${d.risk?.score||0})`;
      updateRisk(d.risk?.score||0,d.risk?.level||'LOW',d.risk?.reasons||[]);
      log(`Validated · uses left: ${d.uses_left}`,'ok');
    }else{
      blocked++;animate('s-blocked',blocked);
      out.className='out-box err';
      const detail=d.detail||d;
      out.textContent=`✗ Token INVALID\n${JSON.stringify(detail,null,2)}`;
      // Check if high risk and show telegram feed
      if(detail?.reason==='high_risk_detected'&&detail?.risk){
        const risk=detail.risk;
        updateRisk(risk.score,risk.level,risk.reasons||[]);
        addTelegramAlert('Anomaly HIGH Risk','127.0.0.1',risk.score,risk.reasons||[]);
      }
      log('Validation rejected','err');
    }
  }catch{out.className='out-box err';out.textContent='✗ Connection error';}
}
